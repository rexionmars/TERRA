% TERRA -- The method and the limit

% English edition of `../livro.tex`. Same product, same structure, same cover.
% What changes is the language, and three things that follow from it: `babel`
% is loaded for English, siunitx marks decimals with a point, and the
% bibliography heading reads "References".
%
% This is not a second book. Chapter for chapter and section for section it
% carries the same argument as the Portuguese edition, and a claim added on one
% side is added on the other. Where the two disagree, the Portuguese edition is
% the one that was written first and the one to check against the code.
%
% Compile from the repository root:
%   make book-en
% Or from here:
%   TEXINPUTS="$(cd ../.. && pwd):$TEXINPUTS" latexmk -pdf book.tex
% latexmk lives in /Library/TeX/texbin, outside the default PATH.
%
% After compiling, measure the cover:
%   ../../../../.venv/bin/python ../../scripts/verifica_capa.py book.pdf
%
% FIGURES COME FROM `../figuras/en/`, produced by `figuras.py --en`. Same
% script and same measurement as the Portuguese edition; what differs is the
% string table and the decimal marker. Regenerate both with:
%   ../../../../.venv/bin/python ../figuras/figuras.py
%   ../../../../.venv/bin/python ../figuras/figuras.py --en
%
% Class `book`, for the two reasons the Portuguese edition gives: `\part`
% groups fifteen chapters into a contents page that fits on one sheet, and
% `\frontmatter` numbers the front matter in roman so that adding a page to the
% contents does not renumber the body.
% =====================================================================

\documentclass[11pt,a4paper,oneside]{book}

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[english]{babel}

% MARGIN OF 2.0 CM, the same as `guia.tex` and as the Portuguese edition, and
% not the 3.5 cm of the scientific template. The three reasons are recorded in
% `../livro.tex`: the two manuals are one pair and must measure alike, the
% screenshot pipeline targets a 17.0 cm block, and the three-column tables
% carry the limit in the third column, which breaks every two words at 14 cm.
\usepackage[margin=2.0cm]{geometry}
\usepackage{terra}

% OPEN LEADING, because of the margin. At 2.0 cm the block is 17.0 cm wide and
% the line runs to about 87 characters, against the 66 that typography treats
% as comfortable. What tires the eye on a long line is the return, not the
% length: opening the vertical gives back the track the length took away.
\linespread{1.10}

\usepackage{microtype}
\usepackage{siunitx}
\usepackage{tocloft}
% The provenance table in appendix B: `longtable` because it crosses pages, and
% `pdflscape` because the file-and-symbol column neither fits nor hyphenates at
% 17 cm. `pdflscape` rotates the page in the PDF as well as the content, so it
% reads without turning the head in a screen reader.
\usepackage{longtable}
\usepackage{pdflscape}

% Leader dots on the chapter line too, which class `book` does not do. Without
% them the chapter page number floats loose on the far side of the sheet. The
% gap between chapters drops from 1 em to 0.4, or the contents page overflows
% without becoming more legible.
\renewcommand{\cftchapleader}{\cftdotfill{\cftdotsep}}
\setlength{\cftbeforechapskip}{.4em}

% Width of the section-number box in the contents. The 2.3 em default fits
% "9.9" and overflows on "16.10": two chapter digits with two section digits.
\setlength{\cftsecnumwidth}{2.9em}

% DECIMAL POINT, and this is the one numeric convention that separates the two
% editions. `writing.md` §11.3: the Portuguese documents use a decimal comma,
% the English ones must not. The thousands separator is a comma, which is what
% the reader of an English table expects.
\sisetup{output-decimal-marker={.},group-separator={,},
         group-minimum-digits=4,separate-uncertainty=true}

% LEFT-ALIGNED COLUMN, inherited from the user guide at the same width and for
% the same reason. The tables here have narrow columns and `p{}` justifies: at
% that width justification opens holes between words. The `plus 2em` instead of
% the `plus 1fil` of \raggedright preserves the diagnostic: with infinite
% stretch on the right TeX never finds a line too wide, and a cell holding a
% monospaced file path, which does not hyphenate, spills over the neighbouring
% column without a single `Overfull` line in the log.
\newcolumntype{L}[1]{>{\raggedright\arraybackslash\rightskip=0pt plus 2em\relax\hbadness=10000\relax}p{#1}}

% This book quotes interface labels in \texttt, and none of them hyphenate:
% `Class separability` and `Domain shift` in one sentence are two rigid blocks.
% When they fall together at the end of a paragraph, TeX prefers to overrun the
% margin to setting a loose line. The emergencystretch buys one extra pass with
% additional stretch, and only in the paragraphs that would otherwise overrun.
\setlength{\emergencystretch}{2em}

% A message or label quoted from the screen word for word, distinct from the
% \texttt that names a control. Same convention as the user guide.
\newcommand{\msg}[1]{``\textit{#1}''}

% FILE PATHS in \path{...}, not in \texttt{...}. The `url` package, already
% loaded by hyperref in terra.sty, allows a line break after "/" and handles
% "_" without escaping. In bare \texttt,
% `docs/TERRA/notas/triagem-estudos-features.md` is a rigid 42-character block.
%
% `\urlstyle{tt}` so that the path comes out in the same monospace as the
% labels: without it `url` sets in the class `\ttfamily`, which is not the
% package plxMono, and two paths on one page came out in two fonts.
\usepackage{url}
\urlstyle{tt}

% FIGURES COME FROM `../figuras/en/`, which `figuras.py --en` writes. One
% drawing, two languages: the same function draws both editions, and what
% changes is the string table and the decimal marker. Duplicating the drawing
% code is what would let the two versions drift on the first correction someone
% forgot to apply on both sides.
%
% The path is written out in the chapter, `figuras/en/fig-x.pdf`, rather than
% hidden in the search path. A `\graphicspath` prefix cannot rewrite the middle
% of a relative name, and spelling it out means the source says which artefact
% is drawn.
\graphicspath{{../}}

% THE COVER IS THE ONE FROM `templates/template-cientifico.tex`, unchanged:
% same photograph, same darkening, same mark and same kicker. The 0.60 is not
% taste, it is the measured minimum for this photograph in this layout, and
% re-measuring it for a second edition of the same book would redo a
% measurement that already exists. The three footer columns are the only things
% that change, because they name THIS document.
%
% Measure again after compiling:
%   ../../../../.venv/bin/python ../../scripts/verifica_capa.py book.pdf
\setcapaimagem{../../images/capas/delta-a4.jpg}
\setcapaescuro{0.60}
% The emblem beside the wordmark, as in `guia.tex` and in both templates: the
% TERRA TitleBar puts the mark on the left and the word TERRA beside it, and a
% cover that repeats that composition makes the document and the product read
% as one thing. 2.9 cm against the roughly 2.2 cm cap height of the wordmark at
% 86 pt, raised by (2.2-2.9)/2 to centre on it; below about 2.8 cm the text in
% the ring stops resolving and the emblem becomes a dark dot.
\setcapamarca{%
  \raisebox{-0.35cm}{\includegraphics[height=2.9cm]{../../images/TERRA_EO_Logo.png}}%
  \hspace{0.46cm}TERRA}
% THE KICKER NUMBERS THE DOCUMENT, not the software. Same decision as
% `guia.tex`, and for the same reason: the TERRA version changes with every
% release and would leave the cover out of date without a line of the text
% having changed. The book's own number changes when the book changes, which is
% what a number on a cover promises.
%
% The TERRA version the text describes is still declared, in the place where it
% decides something: the first field of the preface.
\setcapachapeu{v1}
% THE THREE COLUMNS SAY WHAT THE DOCUMENT ANSWERS, not what TERRA is. Whoever
% picks this book up already uses the application. They are the three questions
% of the preface, in the order the book answers them.
%
% LABEL AS NOUN PHRASE, not as clause. A journal cover names the subject,
% "Ocean circulation", and does not ask where the water goes.
%
% The column is 5.3 cm, which gives about 30 characters per line at 9.5 pt. A
% body longer than two or three lines hyphenates, and a word split on a cover
% does not read as a line break, it reads as an error.
\setcapacolunas
  {The model input}{The scenes, the bands, and the space it was trained in}
  {The disagreement}{Quantity against allocation, and agreement by block}
  {The limit of the product}{The grid of the map, against what the source resolves}
